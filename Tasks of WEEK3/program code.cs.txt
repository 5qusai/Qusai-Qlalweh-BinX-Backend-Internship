public class EmpDept


{
    public string EmpName;
    public string DeptName;

    public EmpDept(string empN,string deptN)
    {
        EmpName=empN;
        DeptName = deptN;
    }
}
public class Department
{

    public string Name;
    public string ID;

    public Department()
    {
        
    }
    public Department(string id,string name)
    {
        Name = name;
        ID = id;
    }

}

public class Employee
{

    public string Name;
    public string ID;
    public string DepID;
    public List<string> Skills;

    public Employee()
    {
        
    }
    public Employee(string name,string id,string depid)
    {
        Name = name;
        ID = id;
        DepID = depid;
        Skills = new List<string>();
    }
    public Employee(string name, string id, string depid,List<string> list)
    {
        Name = name;
        ID = id;
        DepID = depid;
        Skills = list;

    }
}
